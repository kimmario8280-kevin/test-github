class InformationStrip(BaseModel):
    """추출된 정보 조각 — 내용/출처/관련성/충실성"""

loader = PyPDFLoader(pdf_file_civil, mode='single')

# 헤더/푸터 제거: "법제처  N  국가법령정보센터 / 민법"
# (PDF 텍스트 추출 시 '국가법령정보센터'가 줄바꿈으로 갈라지는 경우까지 허용)
text_for_delete_civil = r"법제처\s+\d+\s+국가법령정보센\s*터\s*\n?\s*민법"
civil_law_text = "\n".join(

def parse_law_v3(law_text):
    """'편 > 장 > 절 > (관) > 조' 다단계 구조 법령 파서 (민법 대응).

    - parse_law(장 기준), parse_law_v2(장 없음 대응) 의 확장판
    - 반환: {"서문": str, "편": {편제목: {장제목: {절제목: [조문...]}}}, "부칙": str}
    - 절이 없는 장은 절 키를 "_" 로 통일하여 구조 일관성 유지
    - 부칙이 없으면 빈 문자열
    """
    # 부칙 분리 ('부칙' 단독 라인 기준, 없으면 빈 문자열)
    appendix = ""
    m = re.search(r'(?m)^\s*부칙\s*(?:<[^>]*>)?\s*$', law_text)
    if m:
        appendix = law_text[m.start():].strip()
        law_text = law_text[:m.start()]

    # 조 분리 패턴 — 기존 parse_law 와 동일한 아이디어
    article_pattern = r'(제\d+조(?:의\d+)?\s*\([^)]+\).*?)(?=제\d+조(?:의\d+)?\s*\([^)]+\)|$)'

    def split_articles(text):
        # '제N관 ...' 소제목 라인은 조문 본문을 오염시키지 않도록 제거
        text = re.sub(r'(?m)^\s*제\d+관\s+\S[^\n]*$', '', text)
        return [a.strip() for a in re.findall(article_pattern, text, re.DOTALL)]

    # 편 단위 분리: 줄 전체가 '제N편 제목' 인 라인 기준
    # ('제1편의 규정...' 같은 본문 내 참조는 편 번호 뒤 공백 필수 조건으로 배제)
    parts = re.split(r'(?m)^\s*(제\d+편\s+\S[^\n]*?)\s*$', law_text)
    parsed = {"서문": parts[0].strip(), "편": {}, "부칙": appendix}

    # parts = [서문, 편제목1, 편본문1, 편제목2, 편본문2, ...]
    for i in range(1, len(parts), 2):
        pyeon_title = re.sub(r'\s+', ' ', parts[i]).strip()
        pyeon_body = parts[i + 1]
        parsed["편"][pyeon_title] = {}

        # 장 단위 분리 (장이 없는 편은 장 키 "_")
        jang_parts = re.split(r'(?m)^\s*(제\d+장\s+\S[^\n]*?)\s*$', pyeon_body)
        if len(jang_parts) == 1:
            parsed["편"][pyeon_title]["_"] = {"_": split_articles(pyeon_body)}
            continue

        for j in range(1, len(jang_parts), 2):
            jang_title = re.sub(r'\s+', ' ', jang_parts[j]).strip()
            jang_body = jang_parts[j + 1]
            parsed["편"][pyeon_title][jang_title] = {}

            # 절 단위 분리 (절이 없는 장은 절 키 "_")
            jeol_parts = re.split(r'(?m)^\s*(제\d+절\s+\S[^\n]*?)\s*$', jang_body)
            if len(jeol_parts) == 1:
                parsed["편"][pyeon_title][jang_title]["_"] = split_articles(jang_body)
            else:
                # 절 시작 전에 장 직속 조문이 있으면 "_" 절로 수용
                head_articles = split_articles(jeol_parts[0])
                if head_articles:
                    parsed["편"][pyeon_title][jang_title]["_"] = head_articles
                for k in range(1, len(jeol_parts), 2):
                    jeol_title = re.sub(r'\s+', ' ', jeol_parts[k]).strip()
                    parsed["편"][pyeon_title][jang_title][jeol_title] = split_articles(jeol_parts[k + 1])

    return parsed


def flatten_civil_law(parsed):
    """parse_law_v3 결과를 조문 단위 dict 리스트로 평탄화."""
    flat = []
    for pyeon, jangs in parsed["편"].items():
        for jang, jeols in jangs.items():
            for jeol, articles in jeols.items():
                for article in articles:
                    no_m = re.match(r'(제\d+조(?:의\d+)?)', article)
                    title_m = re.match(r'제\d+조(?:의\d+)?\s*\(([^)]+)\)', article)
                    flat.append({
                        "part": pyeon,
                        "chapter": jang,
                        "section": jeol,
                        "article_no": no_m.group(1) if no_m else "",
                        "article_title": f"({title_m.group(1)})" if title_m else "",
                        "article": article,
                    })
    return flat

from langchain.tools import tool
from langchain_core.documents import Document
from typing import List


@tool
def civil_law_search(query: str) -> List[Document]:
    """민법 법률 조항을 검색합니다. (계약, 채권·채무, 소유권·물권, 불법행위, 손해배상, 혼인·이혼·친권, 상속·유언 등)"""
    docs = civil_db_retriever.invoke(query)

    if len(docs) > 0:
        return docs

    return [Document(page_content="**민법 검색**에서 관련 정보를 찾을 수 없습니다.", metadata={"source": "data/civil_law.pdf", "name": "민법"})]


@tool
def civil_article_lookup(article_no: str) -> List[Document]:
    """민법 조문 번호(예: '제839조의2', '제5조')로 해당 조문 원문을 정확히 조회합니다."""
    # 입력 정규화: 공백 제거, '제'/'조' 보정 ("839" → "제839조")
    no = article_no.strip().replace(" ", "")
    if not no.startswith("제"):
        no = "제" + no
    if not (no.endswith("조") or "조의" in no):
        no = no + "조"

    result = civil_db.get(where={"article_no": no})
    docs = [
        Document(page_content=content, metadata=meta)
        for content, meta in zip(result["documents"], result["metadatas"])
    ]

    if len(docs) > 0:
        return docs

    return [Document(page_content=f"**민법 조문 조회**에서 '{article_no}' 조문을 찾을 수 없습니다.", metadata={"source": "data/civil_law.pdf", "name": "민법"})]

# ── trace 자동 기록 래퍼 ──
def traced(node_name):
    """노드 실행 시간을 재고 trace 에 {node, elapsed, in, out_summary} 를 기록하는 데코레이터"""
    def decorator(fn):
        @functools.wraps(fn)
        def wrapper(state, config=None):
            t0 = time.perf_counter()
            out = fn(state, config) or {}
            elapsed = round(time.perf_counter() - t0, 3)
            in_summary = (state.get("rewritten_query") or state.get("question", ""))[:60]
            out_summary = {
                k: (len(v) if isinstance(v, list) else (v if isinstance(v, (int, float)) else str(v)[:60]))
                for k, v in out.items() if k != "trace"
            }
            out.setdefault("trace", []).append(
                {"node": node_name, "elapsed": elapsed, "in": in_summary, "out_summary": out_summary}
            )
            return out
        return wrapper
    return decorator

@traced("retrieve_civil")
def retrieve_civil(state: CivilRagState, config=None):
    """3. 리트리버 선택형 검색 (basic/mmr/rerank 를 config 로 전환)"""
    conf = (config or {}).get("configurable", {})
    retriever_name = conf.get("retriever", "rerank")
    top_k = conf.get("top_k", 5)

    # 재작성 후에는 편 필터를 해제하여 검색 범위를 넓힘
    part_filter = state.get("part_filter") if state.get("num_rewrites", 0) == 0 else None
    query = state.get("rewritten_query") or state["question"]
    queries = [query] + state.get("sub_questions", [])

    retriever = _build_civil_retriever(retriever_name, part_filter, top_k)
    seen, docs = set(), []
    for q in queries:
        for d in retriever.invoke(q):
            key = (d.metadata.get("article_no"), d.metadata.get("chunk_index"))
            if key not in seen:
                seen.add(key)
                docs.append(d)

    # 편 필터 검색이 0건이면 필터를 해제하고 즉시 재검색 (오분류 대비 폴백)
    if not docs and part_filter:
        print(f"  편 필터({part_filter}) 검색 0건 → 필터 해제 재검색")
        part_filter = None
        retriever = _build_civil_retriever(retriever_name, None, top_k)
        for q in queries:
            for d in retriever.invoke(q):
                key = (d.metadata.get("article_no"), d.metadata.get("chunk_index"))
                if key not in seen:
                    seen.add(key)
                    docs.append(d)

    docs = docs[:8]  # 병렬 평가 LLM 호출 수 상한
    print(f"---민법 검색({retriever_name}, filter={part_filter}): {len(docs)}개 문서---")
    return {"documents": docs}

@traced("aggregate_results")
def aggregate_civil_results(state: CivilRagState, config=None):
    """6. Map 결과 병합 + query_relevance 산출"""
    strips = state.get("extracted_info", [])
    print(f"---민법 정보 집계: 총 {len(strips)}개 조각---")

    # 중복 제거 → 품질순 정렬 → 상위 10개 (기존 개인정보보호법 aggregate 와 동일 아이디어)
    unique_strips, seen_content = [], set()
    for strip in strips:
        if strip.content not in seen_content:
            unique_strips.append(strip)
            seen_content.add(strip.content)
    sorted_strips = sorted(
        unique_strips,
        key=lambda x: (x.relevance_score + x.faithfulness_score) / 2,
        reverse=True,
    )
    top_strips = sorted_strips[:10]

    # 검색 관련성 = 상위 조각들의 평균 관련성 (조각이 없으면 0)
    query_relevance = (
        sum(s.relevance_score for s in top_strips) / len(top_strips) if top_strips else 0.0
    )
    print(f"  query_relevance: {query_relevance:.2f}")
    return {"processed_info": top_strips, "query_relevance": round(query_relevance, 3)}

from typing import Dict

NODE_REGISTRY: Dict[str, dict] = {}


def register_node(name, group, desc, optional=True):
    """노드 함수를 NODE_REGISTRY 에 등록하는 데코레이터.

    모든 노드는 (state, config) -> dict 시그니처를 지키며,
    trace 자동 기록은 STEP 5 의 traced 래퍼가 담당합니다.
    """
    def decorator(fn):
        NODE_REGISTRY[name] = {"fn": fn, "group": group, "desc": desc, "optional": optional}
        return fn
    return decorator

from dataclasses import dataclass, field
from typing import List, Literal

from langgraph.graph import StateGraph, START, END
from langgraph.types import Send


@dataclass
class ExperimentConfig:
    """민법 RAG 실험 설정 — 노드 조합/리트리버/임계값/프롬프트 변형을 전환"""
    name: str
    retriever: Literal["basic", "mmr", "rerank"] = "rerank"
    top_k: int = 5
    enabled_nodes: List[str] = field(default_factory=lambda: list(ALL_OPTIONAL_NODES))
    llm_model: str = "gpt-4.1-mini"
    temperature: float = 0.0
    max_rewrites: int = 2
    relevance_threshold: float = 0.6
    answer_threshold: float = 0.7
    prompt_variant: Literal["basic", "structured", "citation_heavy"] = "structured"

    def to_runnable_config(self):
        """LangGraph invoke 용 config 변환 (기본 모델이면 전역 llm 재사용)"""
        return {
            "recursion_limit": 60,
            "configurable": {
                "retriever": self.retriever,
                "top_k": self.top_k,
                "llm_model": None if (self.llm_model == "gpt-4.1-mini" and self.temperature == 0.0) else self.llm_model,
                "temperature": self.temperature,
                "max_rewrites": self.max_rewrites,
                "relevance_threshold": self.relevance_threshold,
                "answer_threshold": self.answer_threshold,
                "prompt_variant": self.prompt_variant,
            },
        }


# ── 글루(어댑터) 노드: optional 노드가 꺼져도 그래프가 깨지지 않게 연결 ──
@traced("docs_passthrough")
def _civil_docs_passthrough(state: CivilRagState, config=None):
    """grade_documents 가 꺼진 경우: documents 를 평가 없이 graded_docs 로 전달"""
    return {"graded_docs": state.get("documents", [])}


@traced("docs_to_info")
def _civil_docs_to_info(state: CivilRagState, config=None):
    """refine_knowledge 가 꺼진 경우: 문서를 LLM 호출 없이 InformationStrip 으로 변환"""
    docs = state.get("graded_docs") or state.get("documents", [])
    unique, seen = [], set()
    for d in docs:
        key = (d.metadata.get("article_no"), d.metadata.get("chunk_index"))
        if key not in seen:
            seen.add(key)
            unique.append(d)
    strips = [
        InformationStrip(
            content=d.page_content[:1500],
            source=f"{d.metadata.get('name', '민법')} {d.metadata.get('article_no', '')}".strip(),
            relevance_score=float(d.metadata.get("relevance_score", 0.75)),
            faithfulness_score=0.75,
        )
        for d in unique[:10]
    ]
    return {"extracted_info": strips}


def _make_grade_workers(fallback_node):
    """검색 결과 → grade_documents 병렬 워커 생성 (결과 없으면 fallback 으로)"""
    def _workers(state: CivilRagState):
        if not state["documents"]:
            return fallback_node
        question = state.get("rewritten_query") or state["question"]
        return [
            Send("grade_documents", {
                "document": doc.page_content,
                "doc_metadata": doc.metadata,
                "question": question,
            })
            for doc in state["documents"]
        ]
    return _workers


def build_civil_graph(config: ExperimentConfig):
    """ExperimentConfig 에 따라 민법 RAG 그래프를 동적으로 구성/컴파일"""
    enabled = set(config.enabled_nodes) & set(ALL_OPTIONAL_NODES)
    g = StateGraph(CivilRagState)

    # ── 필수 노드 ──
    g.add_node("retrieve_civil", retrieve_civil)
    g.add_node("aggregate_results", aggregate_civil_results)
    g.add_node("generate_civil_answer", generate_civil_answer)

    # ── 전처리 체인: START → (analyze) → (decompose) → retrieve ──
    pre_chain = [n for n in ["analyze_civil_question", "decompose_question"] if n in enabled]
    for n in pre_chain:
        g.add_node(n, NODE_REGISTRY[n]["fn"])
    chain = [START] + pre_chain + ["retrieve_civil"]
    for a, b in zip(chain, chain[1:]):
        g.add_edge(a, b)

    # ── 평가/정제 체인: retrieve → (grade∥) → (refine | docs_to_info) → aggregate ──
    use_grade = "grade_documents" in enabled
    use_refine = "refine_knowledge" in enabled

    if use_refine:
        g.add_node("refine_knowledge", NODE_REGISTRY["refine_knowledge"]["fn"])
        after_docs = "refine_knowledge"
    else:
        g.add_node("docs_to_info", _civil_docs_to_info)
        after_docs = "docs_to_info"

    if use_grade:
        g.add_node("grade_documents", NODE_REGISTRY["grade_documents"]["fn"])
        g.add_conditional_edges("retrieve_civil", _make_grade_workers(after_docs),
                                ["grade_documents", after_docs])
        g.add_edge("grade_documents", after_docs)
    elif use_refine:
        # 평가 없이 정제만 하는 경우: documents 를 graded_docs 로 넘기는 글루 노드
        g.add_node("docs_passthrough", _civil_docs_passthrough)
        g.add_edge("retrieve_civil", "docs_passthrough")
        g.add_edge("docs_passthrough", after_docs)
    else:
        g.add_edge("retrieve_civil", after_docs)
    g.add_edge(after_docs, "aggregate_results")

    # ── 교정 분기: aggregate → rewrite / web_fallback / generate ──
    use_rewrite = "rewrite_query" in enabled
    use_web = "web_fallback" in enabled
    if use_rewrite:
        g.add_node("rewrite_query", NODE_REGISTRY["rewrite_query"]["fn"])
        g.add_edge("rewrite_query", "retrieve_civil")
    if use_web:
        g.add_node("web_fallback", NODE_REGISTRY["web_fallback"]["fn"])
        g.add_edge("web_fallback", "generate_civil_answer")

    def route_aggregate(state, config_rt=None):
        if state.get("query_relevance", 0.0) < config.relevance_threshold:
            if use_rewrite and state.get("num_rewrites", 0) < config.max_rewrites:
                return "rewrite_query"
            if use_web and not state.get("web_docs"):
                return "web_fallback"
        return "generate_civil_answer"

    agg_map = {"generate_civil_answer": "generate_civil_answer"}
    if use_rewrite:
        agg_map["rewrite_query"] = "rewrite_query"
    if use_web:
        agg_map["web_fallback"] = "web_fallback"
    g.add_conditional_edges("aggregate_results", route_aggregate, agg_map)

    # ── 평가 체인: generate → (grade_hallucination) → (grade_answer ⟲) → (verify_citations) → END ──
    eval_chain = [n for n in ["grade_hallucination", "grade_answer", "verify_citations"] if n in enabled]
    for n in eval_chain:
        g.add_node(n, NODE_REGISTRY[n]["fn"])

    def _make_route_grade(next_node):
        """grade_answer 점수 미달 시 재작성 루프, 아니면 다음 노드로"""
        def route_grade(state, config_rt=None):
            if state.get("answer_score", 0.0) < config.answer_threshold and state.get("num_generations", 0) < 3:
                return "rewrite_query"
            return "next"
        return route_grade, {"rewrite_query": "rewrite_query", "next": next_node}

    seq = ["generate_civil_answer"] + eval_chain + [END]
    for a, b in zip(seq, seq[1:]):
        if a == "grade_answer" and use_rewrite:
            router, path_map = _make_route_grade(b)
            g.add_conditional_edges("grade_answer", router, path_map)
        else:
            g.add_edge(a, b)

    compiled = g.compile()
    compiled.get_graph()  # 그래프 유효성 검사 (구성 오류 시 예외 발생)
    return compiled

def estimate_llm_calls(config: ExperimentConfig, n_questions: int) -> int:
    """실행 전 예상 LLM 호출 수 (재작성 루프 제외한 1회 통과 기준)"""
    enabled = set(config.enabled_nodes)
    per_run = 1  # generate
    per_run += len(enabled & {"analyze_civil_question", "decompose_question",
                              "refine_knowledge", "grade_hallucination", "grade_answer"})
    if "grade_documents" in enabled:
        per_run += 8  # 병렬 문서 평가 상한
    return per_run * n_questions


def run_experiment(configs, questions) -> pd.DataFrame:
    """각 (config, question) 조합을 실행하고 지표를 수집하여 DataFrame 으로 반환"""
    _os.makedirs(EXPERIMENT_DIR, exist_ok=True)
    rows = []

    for cfg in configs:
        try:
            graph = build_civil_graph(cfg)
        except Exception as e:
            rows.append({"config": cfg.name, "question": "-", "error": f"그래프 빌드 실패: {e}"})
            continue

        for q in questions:
            row = {"config": cfg.name, "question": q, "error": ""}
            t0 = time.perf_counter()
            try:
                with get_openai_callback() as cb:
                    result = graph.invoke({"question": q}, config=cfg.to_runnable_config())
                row.update({
                    "answer": result.get("node_answer", ""),
                    "elapsed_s": round(time.perf_counter() - t0, 2),
                    "llm_calls": cb.successful_requests,
                    "total_tokens": cb.total_tokens,
                    "cost_usd": round(cb.total_cost, 5),
                    "num_docs": len(result.get("documents", [])),
                    "hallucination_score": result.get("hallucination_score"),
                    "answer_score": result.get("answer_score"),
                    "citation_valid_ratio": result.get("citation_valid_ratio"),
                    "num_rewrites": result.get("num_rewrites", 0),
                    "node_times": {t["node"]: t["elapsed"] for t in result.get("trace", [])},
                })
            except Exception as e:
                # 실패한 실행은 error 에 기록하고 계속 진행
                row["error"] = str(e)[:200]
                row["elapsed_s"] = round(time.perf_counter() - t0, 2)
            rows.append(row)
            print(f"  [{cfg.name:10s}] {q[:28]}… → {row.get('elapsed_s', '?')}s "
                  f"(오류: {row['error'] or '없음'})")

    df = pd.DataFrame(rows)
    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    csv_path = f"{EXPERIMENT_DIR}/results_{ts}.csv"
    df.drop(columns=["node_times"], errors="ignore").to_csv(csv_path, index=False, encoding="utf-8-sig")
    with open(f"{EXPERIMENT_DIR}/results_{ts}.json", "w", encoding="utf-8") as f:
        _json.dump(rows, f, ensure_ascii=False, indent=1, default=str)
    print(f"\n결과 저장: {csv_path} / .json")
    return df

def merge_agent_responses(left: dict, right: dict) -> dict:
    """에이전트 응답 병합 커스텀 리듀서 (병렬 실행 지원)"""
    if left is None:
        left = {}
    if right is None:
        right = {}
    merged = left.copy()
    merged.update(right)
    return merged


class ResearchAgentState(TypedDict):
    """질문 라우팅 및 다중 에이전트 조합 시스템의 상태 정의"""
    question: str
    answers: Annotated[List[str], add]
    agent_responses: Annotated[dict, merge_agent_responses]
    final_answer: str

def _lite_law_node(agent_key, law_label, search_tool):
    """간이 법률 에이전트: 검색 → 생성 1-패스 (원본 노트북은 CRAG 서브그래프 사용)"""
    def _node(state):
        print(f"--- {law_label} 간이 에이전트 시작 ---")
        question = state["question"]
        feedback = state.get("user_feedback") or ""
        try:
            docs = search_tool.invoke(question)
            context = "\n\n".join(d.page_content for d in docs[:4])
        except Exception as e:
            context = f"(검색 실패: {e})"
        prompt = ChatPromptTemplate.from_messages([
            ("system", f"당신은 {law_label} 전문가입니다. 주어진 법률 조항을 근거로 질문에 답하세요. "
                       "관련 조항 번호를 반드시 인용하고, 근거가 없으면 그 한계를 밝히세요."),
            ("human", "<질문>\n{question}\n</질문>\n<법률 조항>\n{context}\n</법률 조항>{feedback}"),
        ])
        answer = llm.invoke(prompt.format(
            question=question, context=context,
            feedback=f"\n<사용자 피드백>\n{feedback}\n</사용자 피드백>" if feedback else "",
        ))
        current = state.get("agent_responses", {}).copy()
        current[agent_key] = answer.content
        return {"answers": [answer.content], "agent_responses": current}
    return _node

import pandas as pd
from langgraph.types import Command

TEST_QUERIES = [
    "개인정보를 제3자에게 제공할 때 필요한 동의 요건은?",
    "회사에서 직원의 개인정보를 수집할 때 근로기준법과 개인정보보호법 상 주의사항은?",
    "2024년 개인정보보호법 개정 내용과 기존 조항의 차이점은?",
    "타인의 물건을 선의취득하려면 어떤 요건이 필요한가요?",
    "임대인이 보증금을 반환하지 않을 때 민법상 구제수단은?",
]

test_rows = []
for tq in TEST_QUERIES:
    print("=" * 90)
    print("질문:", tq)
    t_config = {"configurable": {"thread_id": str(uuid.uuid4())}, "recursion_limit": 80}
    t_init = {
        "question": tq, "answers": [], "agent_responses": {}, "final_answer": "",
        "datasources": [], "evaluation_report": {}, "iteration_count": 0,
        "user_decision": "", "user_feedback": "", "node_scores": {},
        "experiment_config": None, "feedback_history": [],
    }
    t0 = time.perf_counter()
    try:
        interrupted = False
        for chunk in legal_rag_agent_v2.stream(t_init, config=t_config):
            if "__interrupt__" in chunk:
                interrupted = True
                break
        if interrupted:
            # HITL 자동 승인으로 재개
            t_result = legal_rag_agent_v2.invoke(Command(resume="y"), config=t_config)
        else:
            t_result = legal_rag_agent_v2.get_state(t_config).values

        t_answer = t_result.get("final_answer", "")
        t_report = t_result.get("evaluation_report", {}) or {}
        t_citations = list(dict.fromkeys(re.findall(r"제\d+조(?:의\d+)?", t_answer)))
        test_rows.append({
            "질문": tq[:24] + "…",
            "선택된 도구": ",".join(t_result.get("datasources", [])),
            "인용 조항": "; ".join(t_citations[:5]),
            "정확성": t_report.get("accuracy_score"),
            "완전성": t_report.get("completeness_score"),
            "총점/60": t_report.get("total_score"),
            "소요(s)": round(time.perf_counter() - t0, 1),
            "오류": "",
        })
    except Exception as e:
        test_rows.append({"질문": tq[:24] + "…", "선택된 도구": "-", "인용 조항": "-",
                          "정확성": None, "완전성": None, "총점/60": None,
                          "소요(s)": round(time.perf_counter() - t0, 1), "오류": str(e)[:120]})
    print("  →", test_rows[-1])
